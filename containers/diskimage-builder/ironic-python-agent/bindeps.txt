# IPA ramdisk package dependencies for container-based image builds.
#
# This file lists all RPM packages required to build an
# ironic-python-agent ramdisk image. Packages are derived from:
#   - IPA source code analysis (subprocess/execute calls to system binaries)
#   - diskimage-builder element package-installs.yaml and pkg-map files
#   - ironic-python-agent-builder element requirements
#
# Packages are listed with their RHEL/CentOS package names.
# Comments indicate the source of each requirement.

# === Disk partitioning and partition table tools ===
# IPA source: disk_utils.py, disk_partitioner.py, partition_utils.py, hardware.py
# DIB element: ironic-python-agent-ramdisk-base
dosfstools
e2fsprogs
gdisk
parted
util-linux
xfsprogs

# === RAID management ===
# IPA source: hardware.py, raid_utils.py
# DIB element: ironic-python-agent-ramdisk-base
mdadm

# === Storage hardware tools ===
# IPA source: hardware.py (ATA security erase, SMART, NVMe)
# DIB element: ironic-python-agent-ramdisk-base
hdparm
nvme-cli
smartmontools

# === Image conversion ===
# IPA source: qemu_img.py
# DIB element: ironic-python-agent-ramdisk-base (qemu-utils -> qemu-img)
qemu-img

# === EFI and boot management ===
# IPA source: efi_utils.py, extensions/image.py
# DIB element: ironic-python-agent-ramdisk-base
efibootmgr
efivar

# === Network tools ===
# IPA source: agent.py, netutils.py, utils.py
# DIB element: ironic-python-agent-ramdisk-base, dhcp-all-interfaces, simple-init
dhcpcd

net-tools
NetworkManager

coreutils

# === Kernel and boot ===
# Required by DIB select_boot_kernel_initrd (img-functions) which looks for
# /boot/vmlinuz-* and /boot/initramfs-*.img in the target filesystem.
# The base container image has no kernel installed by default.
# A placeholder initramfs is created in the Containerfile since dracut
# cannot run during the container build. The IPA ramdisk build creates
# its own cpio-based initramfs and only copies out the kernel binary.
kernel
kernel-core
kernel-modules
kernel-modules-core
kernel-modules-extra

# === Firmware ===
# DIB element: ironic-python-agent-ramdisk-base (post-install, rh only)
linux-firmware
